   324→   324→    stop();
   325→   325→    const AC = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   326→   326→    const ac = new AC();
   327→   327→    ctxRef.current = ac;
   328→   328→    const master = ac.createGain();

   369→   369→    }
   370→   370→
   371→   371→    if (id === "numbers") {
   372→   372→      nf.frequency.value = 1800;
   373→   373→      ng.gain.value = 0.16;
   374→   374→      noise.connect(nf).connect(ng).connect(master);
   375→   375→      noise.start();

   377→   377→      const beep = (f: number, dur: number, when: number) => {
   378→   378→        const o = ac.createOscillator();
   379→   379→        o.frequency.value = f;
   380→   380→        const g = ac.createGain();
   381→   381→        g.gain.setValueAtTime(0.001, when);
   382→   382→        g.gain.exponentialRampToValueAtTime(0.22, when + 0.02);
   383→   383→        g.gain.exponentialRampToValueAtTime(0.001, when + dur);
   384→   384→        o.connect(g).connect(master);
   385→   385→        o.start(when);
   386→   386→        o.stop(when + dur + 0.05);
   387→   387→      };
   388→   388→      const loop = () => {
   389→   389→        const t0 = ac.currentTime + 0.1;
   390→   390→        [880, 880, 660, 880, 990].forEach((f, i) => beep(f, 0.16, t0 + i * 0.5));
   391→   391→        [660, 660, 660].forEach((f, i) => beep(f, 0.16, t0 + 4 + i * 0.5));
   392→   392→      };
   393→   393→      loop();
   394→   394→      const iv = setInterval(loop, 7000);
   395→   395→      stops.push(() => { noise.stop(); clearInterval(iv); });

   398→   398→    if (id === "message") {

   410→   410→        f.type = "bandpass";
   411→   411→        f.frequency.value = 700;
   412→   412→        const g = ac.createGain();
   413→   413→        g.gain.setValueAtTime(0.0001, when);
   414→   414→        g.gain.exponentialRampToValueAtTime(0.14, when + dur * 0.2);
   415→   415→        g.gain.exponentialRampToValueAtTime(0.0001, when + dur);
   416→   416→        o.connect(f).connect(g).connect(master);
   417→   417→        o.start(when);
   418→   418→        o.stop(when + dur + 0.05);
   419→   419→      };
   420→   420→      const loop = () => {
   421→   421→        let t = ac.currentTime + 0.2;
   422→   422→        const words = [0.5, 0.35, 0.6, 0.3, 0.8, 0.4];
   423→   423→        words.forEach((dur, i) => { say(t, 110 + (i % 3) * 18, dur); t += dur + 0.18; });
   424→   424→      };
   425→   425→      loop();
   426→   426→      const iv = setInterval(loop, 6500);
   427→   427→      stops.push(() => { noise.stop(); clearInterval(iv); });
   428→   428→    }
   429→   429→
   430→   430→    stopRef.current = () => stops.forEach((s) => s());
   431→   431→    setTuned(id);
   432→   432→
   433→   433→    // VU meter
   434→   434→    const data = new Uint8Array(meter.frequencyBinCount);
   435→   435→    const tick = () => {
   436→   436→      meter.getByteFrequencyData(data);
   437→   437→      setLevel(data.reduce((s, v) => s + v, 0) / (data.length * 255));
   438→   438→      if (ctxRef.current === ac) requestAnimationFrame(tick);
   439→   439→    };
   440→   440→    tick();
   441→   441→  };
   442→   442→
   443→   443→  return (
   444→   444→    <div className="w-panel overflow-hidden">
   445→   445→      <div className="flex items-center justify-between border-b border-[var(--w-panel-border)] px-4 py-2.5">
   446→   446→        <p className="flex items-center gap-2 font-[var(--font-ibm-plex-mono)] text-[12px] uppercase tracking-widest opacity-75">
   447→   447→          <Radio className="h-4 w-4" /> The Radio Room
   448→   448→        </p>
   449→   449→        <span className="flex items-center gap-1.5 font-[var(--font-ibm-plex-mono)] text-[10.5px] opacity-50">
   450→   450→          <Signal className="h-3 w-3" style={{ opacity: tuned ? 1 : 0.3 }} /> {tuned ? "signal locked" : "scanning"}
   451→   451→        </span>
   452→   452→      </div>
   453→   453→      <div className="p-5">
   454→   454→        {/* radio panel */}
   455→   455→        <div className="mb-4 flex items-center gap-4 rounded-lg border border-[var(--w-panel-border)] bg-black/30 p-4">
   456→   456→          <div className="flex h-16 flex-1 items-center gap-[3px] overflow-hidden rounded bg-black/60 px-3" aria-hidden>
