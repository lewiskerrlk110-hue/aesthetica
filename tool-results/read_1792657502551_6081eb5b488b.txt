     3→     3→// STUDIO — World editor: rename a world, recolor its theme, assign a
     4→     4→// background media wash (built-ins), build its whole scene (customs), and
     5→     5→// the Type & FX / Position labs — genre font packs, hero title effects,
     6→     6→// hero layout controls — all mirrored live in the preview panel.
     7→     7→import { useEffect, useMemo, useState } from "react";
     8→     8→import { ArrowLeft, RotateCcw, Save, Trash2, Sparkles } from "lucide-react";
     9→     9→import { WORLDS } from "@/lib/aesthetics";
    10→    10→import type { PortalWorldDef, SceneDoc, WorldThemePatch, BgSettings } from "@/lib/studio";
    11→    11→import { GENRE_PACKS, effectClass, fontStack, type GenrePack } from "@/lib/studio";
    12→    12→import { useStudioConfig, publishWorldDraft, type WorldDraftPatch } from "@/lib/worlds-config";
    13→    13→import { SceneEditor } from "./scene-editor";
    14→    14→import { MediaPicker } from "./media-library";
    15→    15→import { FontPicker, EffectPicker, AlignPicker, SliderRow, type AlignValue } from "./design-controls";
    16→    16→import { LivePreview } from "./preview-panel";
    17→    17→import { BLEND_MODES } from "@/lib/studio";
    18→    18→
    19→    19→const THEME_FIELDS: { key: keyof WorldThemePatch; label: string }[] = [
    20→    20→  { key: "bg", label: "Background" },
    21→    21→  { key: "bg2", label: "Background 2" },
    22→    22→  { key: "fg", label: "Text" },
    23→    23→  { key: "accent", label: "Accent" },
    24→    24→  { key: "accent2", label: "Accent 2" },
    25→    25→  { key: "panelBorder", label: "Panel border" },
    26→    26→];
    27→    27→
    28→    28→type EditorTab = "identity" | "type" | "position" | "theme" | "background" | "scene";
    29→    29→
    30→    30→export function WorldEditor({ def, onBack }: { def: PortalWorldDef; onBack: () => void }) {
    31→    31→  const reload = useStudioConfig((s) => s.reload);
    32→    32→  const [name, setName] = useState(def.name);
    33→    33→  const [epithet, setEpithet] = useState(def.epithet);
    34→    34→  const [blurb, setBlurb] = useState(def.portalBlurb);
    35→    35→  const [theme, setTheme] = useState<WorldThemePatch>(def.theme || {});
    36→    36→  const [bgMediaUrl, setBgMediaUrl] = useState<string | null>(def.bgMediaUrl || null);
    37→    37→  const [bgSettings, setBgSettings] = useState<Partial<BgSettings>>({ opacity: 0.5, blend: "normal", fit: "cover", dim: 0.25, ...(def.bgSettings || {}) });
    38→    38→  const [scene, setScene] = useState<SceneDoc>(def.scene || { bg: "#0d0b14", layers: [] });
    39→    39→  const [selectedLayer, setSelectedLayer] = useState<string | null>(null);
    40→    40→  const [pickerOpen, setPickerOpen] = useState(false);
    41→    41→  const [saving, setSaving] = useState(false);
    42→    42→  const [savedAt, setSavedAt] = useState<string | null>(null);
    43→    43→  // custom worlds open straight into the scene builder
    44→    44→  const [tab, setTab] = useState<EditorTab>(() => (def.kind === "custom" ? "scene" : "identity"));
    45→    45→
    46→    46→  // re-sync the whole draft when a different world is opened (render-time adjustment)
    47→    47→  const [lastSlug, setLastSlug] = useState(def.id);
    48→    48→  if (lastSlug !== def.id) {
    49→    49→    setLastSlug(def.id);
    50→    50→    setName(def.name);
    51→    51→    setEpithet(def.epithet);
    52→    52→    setBlurb(def.portalBlurb);
    53→    53→    setTheme(def.theme || {});
    54→    54→    setBgMediaUrl(def.bgMediaUrl || null);
    55→    55→    setBgSettings({ opacity: 0.5, blend: "normal", fit: "cover", dim: 0.25, ...(def.bgSettings || {}) });
    56→    56→    setScene(def.scene || { bg: "#0d0b14", layers: [] });
    57→    57→    setSelectedLayer(null);
    58→    58→    setTab(def.kind === "custom" ? "scene" : "identity");
    59→    59→  }
    60→    60→
    61→    61→  // ——— live preview: stream this world's unsaved edits into every open page ———
    62→    62→  useEffect(() => {
    63→    63→    const t = setTimeout(() => {
    64→    64→      publishWorldDraft(def.id, {
    65→    65→        name,
    66→    66→        epithet,
    67→    67→        portalBlurb: blurb,
    68→    68→        theme,
    69→    69→        bgMediaUrl,
    70→    70→        ...(def.kind === "custom" ? { scene } : { bgSettings: bgSettings as WorldDraftPatch["bgSettings"] }),
    71→    71→      });
    72→    72→    }, 160);
    73→    73→    return () => clearTimeout(t);
    74→    74→  }, [def.id, def.kind, name, epithet, blurb, theme, bgMediaUrl, bgSettings, scene]);
    75→    75→
    76→    76→  // leaving this editor (world switch or unmount) drops its draft
    77→    77→  useEffect(() => {
    78→    78→    const id = def.id;
    79→    79→    return () => publishWorldDraft(id, null);
    80→    80→  }, [def.id]);
    81→    81→
    82→    82→  const dirty = useMemo(() => {
    83→    83→    const bgDefault = { opacity: 0.5, blend: "normal", fit: "cover", dim: 0.25, ...(def.bgSettings || {}) };
    84→    84→    return (
    85→    85→      name !== def.name ||
    86→    86→      epithet !== def.epithet ||
    87→    87→      blurb !== def.portalBlurb ||
    88→    88→      JSON.stringify(theme) !== JSON.stringify(def.theme || {}) ||
    89→    89→      bgMediaUrl !== (def.bgMediaUrl || null) ||
    90→    90→      JSON.stringify(bgSettings) !== JSON.stringify(bgDefault) ||
    91→    91→      (def.kind === "custom" && JSON.stringify(scene) !== JSON.stringify(def.scene || {}))
    92→    92→    );
    93→    93→  }, [def, name, epithet, blurb, theme, bgMediaUrl, bgSettings, scene]);
    94→    94→
    95→    95→  // static defaults for reset hints + effect tinting
    96→    96→  const staticDef = WORLDS.find((w) => w.id === def.id);
    97→    97→  const effAccent = theme.accent || staticDef?.accent || "#ff2e88";
    98→    98→  const effAccent2 = theme.accent2 || staticDef?.accent2 || "#7df2d8";
    99→    99→  const displayStack = fontStack(theme.fontDisplay);
   100→   100→
   101→   101→  const save = async () => {
   102→   102→    setSaving(true);
   103→   103→    const body: Record<string, unknown> = { name, epithet, portalBlurb: blurb };
   104→   104→    if (def.kind === "builtin") {
   105→   105→      body.theme = theme;
   106→   106→      body.bgMediaUrl = bgMediaUrl;
   107→   107→      body.bgSettings = bgSettings;
   108→   108→    } else {
   109→   109→      body.theme = theme;
   110→   110→      body.scene = scene;
   111→   111→    }
   112→   112→    const r = await fetch(`/api/studio/worlds/${def.id}`, {
   113→   113→      method: "PATCH",
   114→   114→      headers: { "Content-Type": "application/json" },
   115→   115→      body: JSON.stringify(body),
   116→   116→    });
   117→   117→    setSaving(false);
   118→   118→    if (r.ok) {
   119→   119→      setSavedAt(new Date().toLocaleTimeString());
   120→   120→      publishWorldDraft(def.id, null); // drop the draft — saved data takes over
   121→   121→      await reload();
   122→   122→    } else {
   123→   123→      const d = await r.json().catch(() => ({}));
   124→   124→      alert(d.error || "Could not save");
   125→   125→    }
   126→   126→  };
   127→   127→
   128→   128→  const resetWorld = async () => {
   129→   129→    if (def.kind === "custom") {
   130→   130→      if (!confirm(`Delete the world "${def.name}" permanently?`)) return;
   131→   131→      await fetch(`/api/studio/worlds/${def.id}`, { method: "DELETE" });
   132→   132→      publishWorldDraft(def.id, null);
   133→   133→      await reload();
   134→   134→      onBack();
   135→   135→      return;
   136→   136→    }
   137→   137→    if (!confirm(`Reset all Studio changes for "${def.name}" back to the original?`)) return;
   138→   138→    await fetch(`/api/studio/worlds/${def.id}`, { method: "DELETE" });
   139→   139→    publishWorldDraft(def.id, null);
   140→   140→    await reload();
   141→   141→    onBack();
   142→   142→  };
   143→   143→
   144→   144→  const setThemeKey = (k: keyof WorldThemePatch, v: string | null) => {
   145→   145→    setTheme((t) => ({ ...t, [k]: v }));
   146→   146→  };
   147→   147→
   148→   148→  const applyPack = (p: GenrePack) =>
   149→   149→    setTheme((t) => ({
   150→   150→      ...t,
   151→   151→      fontDisplay: p.fontDisplay,
   152→   152→      fontBody: p.fontBody,
   153→   153→      epithetFont: p.epithetFont || "",
   154→   154→      titleEffect: p.effect,
   155→   155→    }));
   156→   156→
   157→   157→  const packActive = (p: GenrePack) =>
   158→   158→    theme.fontDisplay === p.fontDisplay &&
   159→   159→    theme.fontBody === p.fontBody &&
   160→   160→    (theme.titleEffect ?? "none") === p.effect;
   161→   161→
   162→   162→  const tabs: { id: EditorTab; label: string; show: boolean }[] = [
   163→   163→    { id: "identity", label: "Identity", show: true },
   164→   164→    { id: "type", label: "Type & FX", show: true },
   165→   165→    { id: "position", label: "Position", show: true },
   166→   166→    { id: "theme", label: "Theme colors", show: true },
   167→   167→    { id: "background", label: "Background media", show: def.kind === "builtin" },
   168→   168→    { id: "scene", label: "Scene builder", show: def.kind === "custom" },
   169→   169→  ];
   170→   170→
   171→   171→  return (
   172→   172→    <div>
   173→   173→      {/* header */}
   174→   174→      <div className="flex flex-wrap items-center gap-3">
   175→   175→        <button onClick={onBack} className="flex items-center gap-1.5 rounded-lg border border-white/12 bg-white/[0.04] px-3 py-1.5 text-[13px] text-white/80 hover:bg-white/10">
   176→   176→          <ArrowLeft className="h-3.5 w-3.5" /> All worlds
   177→   177→        </button>
   178→   178→        <div className="min-w-0">
   179→   179→          <h2 className="truncate text-[17px] font-bold text-white">{def.name}</h2>
   180→   180→          <p className="text-[11.5px] text-white/45">
   181→   181→            {def.kind === "builtin" ? "Built-in world — editable via overrides" : "Custom world — full scene builder"} · /#/world/{def.id}
   182→   182→          </p>
   183→   183→        </div>
   184→   184→        <div className="ml-auto flex items-center gap-2">
   185→   185→          {savedAt && <span className="text-[11.5px] text-[#7df2d8]">Saved {savedAt}</span>}
   186→   186→          {dirty && <span className="text-[11.5px] text-[#ffb347]">Unsaved changes</span>}
   187→   187→          <button
   188→   188→            onClick={resetWorld}
   189→   189→            className="flex items-center gap-1.5 rounded-lg border border-white/12 px-3 py-1.5 text-[12.5px] text-white/60 hover:border-[#ff5c7a]/50 hover:text-[#ff5c7a]"
   190→   190→          >
   191→   191→            <Trash2 className="h-3.5 w-3.5" />
   192→   192→            {def.kind === "custom" ? "Delete world" : "Reset"}
   193→   193→          </button>
   194→   194→          <button
   195→   195→            onClick={save}
   196→   196→            disabled={saving}
   197→   197→            className="flex items-center gap-1.5 rounded-lg bg-gradient-to-r from-[#ff2e88] to-[#7df2d8] px-4 py-1.5 text-[13px] font-semibold text-black disabled:opacity-50"
   198→   198→          >
   199→   199→            <Save className="h-3.5 w-3.5" />
   200→   200→            {saving ? "Saving…" : "Save changes"}
   201→   201→          </button>
   202→   202→        </div>
   203→   203→      </div>
   204→   204→
   205→   205→      {/* tabs */}
   206→   206→      <div className="mt-4 flex gap-1 border-b border-white/10">
   207→   207→        {tabs.filter((t) => t.show).map((t) => (
   208→   208→          <button
   209→   209→            key={t.id}
   210→   210→            onClick={() => setTab(t.id)}
   211→   211→            className={`-mb-px border-b-2 px-3.5 py-2 text-[13px] transition-colors ${
   212→   212→              tab === t.id ? "border-[#ff2e88] font-semibold text-white" : "border-transparent text-white/50 hover:text-white/85"
   213→   213→            }`}
   214→   214→          >
   215→   215→            {t.label}
   216→   216→          </button>
   217→   217→        ))}
   218→   218→      </div>
   219→   219→
   220→   220→      {/* editor + live preview */}
   221→   221→      <div className="mt-5 flex flex-col gap-6 xl:flex-row xl:items-start">
   222→   222→        <div className="min-w-0 flex-1">
   223→   223→          {tab === "identity" && (
   224→   224→            <div className="max-w-xl space-y-4">
   225→   225→              <div>
   226→   226→                <label className="studio-label">World name</label>
   227→   227→                <input className="studio-input" value={name} maxLength={48} onChange={(e) => setName(e.target.value)} />
   228→   228→              </div>
   229→   229→              <div>
   230→   230→                <label className="studio-label">Epithet (the tagline)</label>
   231→   231→                <input className="studio-input" value={epithet} maxLength={80} onChange={(e) => setEpithet(e.target.value)} />
   232→   232→                <p className="mt-1.5 text-[11.5px] text-white/40">
   233→   233→                  Give it a script voice with a genre font — see the Type &amp; FX tab.
   234→   234→                </p>
   235→   235→              </div>
   236→   236→              <div>
   237→   237→                <label className="studio-label">Portal blurb (shown on hover)</label>
   238→   238→                <textarea className="studio-input min-h-[70px]" value={blurb} maxLength={280} onChange={(e) => setBlurb(e.target.value)} />
   239→   239→              </div>
   240→   240→              <p className="text-[12px] leading-relaxed text-white/40">
   241→   241→                The name, epithet and blurb update the portal card on the homepage, the nav dropdown
   242→   242→                and the world page hero. {def.kind === "custom" ? "The slug /#/world/" + def.id + " is permanent." : ""}
   243→   243→              </p>
   244→   244→            </div>
   245→   245→          )}
   246→   246→
   247→   247→          {tab === "type" && (
   248→   248→            <div className="space-y-9">
   249→   249→              {/* ——— genre packs ——— */}
   250→   250→              <section>
   251→   251→                <h3 className="flex items-center gap-2 text-[12px] font-semibold uppercase tracking-[0.3em] text-white/45">
   252→   252→                  <Sparkles className="h-3.5 w-3.5 text-[#ffb347]" /> Genre packs
   253→   253→                </h3>
   254→   254→                <p className="mt-1.5 text-[12px] leading-relaxed text-white/45">
   255→   255→                  One click sets the display font, body font, epithet script and a hero title effect —
   256→   256→                  curated pairings for each internet aesthetic, like an 80s brand mark.
   257→   257→                </p>
   258→   258→                <div className="mt-4 grid gap-3 sm:grid-cols-2 xl:grid-cols-3">
   259→   259→                  {GENRE_PACKS.map((p) => {
   260→   260→                    const tuned = p.worlds.includes(def.id);
   261→   261→                    const eff = effectClass(p.effect);
   262→   262→                    return (
   263→   263→                      <button
   264→   264→                        key={p.key}
   265→   265→                        type="button"
   266→   266→                        onClick={() => applyPack(p)}
   267→   267→                        aria-pressed={packActive(p)}
   268→   268→                        className={`relative rounded-xl border p-4 text-left transition-colors ${
   269→   269→                          packActive(p) ? "border-[#ff2e88] bg-[#ff2e88]/10" : "border-white/12 bg-white/[0.03] hover:bg-white/[0.06]"
   270→   270→                        }`}
   271→   271→                      >
   272→   272→                        {tuned && (
   273→   273→                          <span className="absolute right-3 top-3 rounded-full bg-[#7df2d8]/15 px-2 py-0.5 text-[9.5px] font-bold uppercase tracking-wider text-[#7df2d8]">
   274→   274→                            Tuned
   275→   275→                          </span>
   276→   276→                        )}
   277→   277→                        <p
   278→   278→                          className={`truncate text-[21px] font-black leading-tight ${eff}`}
   279→   279→                          style={{
   280→   280→                            fontFamily: fontStack(p.fontDisplay) ?? undefined,
   281→   281→                            ["--w-accent" as string]: effAccent,
   282→   282→                            ["--w-accent2" as string]: effAccent2,
   283→   283→                            color: eff ? undefined : "#f4f1fa",
   284→   284→                          }}
   285→   285→                        >
   286→   286→                          {p.sample}
   287→   287→                        </p>
   288→   288→                        <p className="mt-2.5 text-[13px] font-semibold text-white">{p.label}</p>
   289→   289→                        <p className="mt-1 text-[11.5px] leading-relaxed text-white/45">{p.blurb}</p>
   290→   290→                        {p.epithetFont && (
   291→   291→                          <p
   292→   292→                            className="mt-1.5 truncate text-[15px] text-[#ff9ecb]"
   293→   293→                            style={{ fontFamily: fontStack(p.epithetFont) ?? undefined }}
   294→   294→                          >
   295→   295→                            {epithet || "script accent"}
   296→   296→                          </p>
   297→   297→                        )}
   298→   298→                      </button>
   299→   299→                    );
   300→   300→                  })}
   301→   301→                </div>
   302→   302→              </section>
   303→   303→
   304→   304→              {/* ——— individual fonts ——— */}
   305→   305→              <section>
   306→   306→                <h3 className="text-[12px] font-semibold uppercase tracking-[0.3em] text-white/45">Fonts</h3>
   307→   307→                <p className="mt-1.5 text-[12px] leading-relaxed text-white/45">
   308→   308→                  Or fine-tune each slot yourself — every specimen renders in the real font.
   309→   309→                </p>
   310→   310→                <div className="mt-4 space-y-6">
   311→   311→                  <div>
   312→   312→                    <label className="studio-label">Display font — headings &amp; hero title</label>
   313→   313→                    <FontPicker
   314→   314→                      value={theme.fontDisplay || ""}
   315→   315→                      onChange={(k) => setThemeKey("fontDisplay", k || null)}
   316→   316→                      defaultLabel="World default"
   317→   317→                    />
   318→   318→                  </div>
   319→   319→                  <div>
   320→   320→                    <label className="studio-label">Body font — paragraphs &amp; buttons</label>
   321→   321→                    <FontPicker
   322→   322→                      value={theme.fontBody || ""}
   323→   323→                      onChange={(k) => setThemeKey("fontBody", k || null)}
   324→   324→                      defaultLabel="World default"
   325→   325→                    />
   326→   326→                  </div>
   327→   327→                  <div>
   328→   328→                    <label className="studio-label">Epithet font — the tagline&apos;s own voice</label>
   329→   329→                    <p className="mb-2 text-[11.5px] text-white/40">
   330→   330→                      A flowing script here (try Mrs Saint Delafield) gives the pink-ink signature look.
   331→   331→                    </p>
   332→   332→                    <FontPicker
   333→   333→                      value={theme.epithetFont || ""}
   334→   334→                      onChange={(k) => setThemeKey("epithetFont", k || null)}
   335→   335→                      defaultLabel="Match the epithet styling"
   336→   336→                    />
   337→   337→                  </div>
   338→   338→                </div>
   339→   339→              </section>
   340→   340→
   341→   341→              {/* ——— title effects ——— */}
   342→   342→              <section>
   343→   343→                <h3 className="text-[12px] font-semibold uppercase tracking-[0.3em] text-white/45">Hero title effect</h3>
   344→   344→                <p className="mt-1.5 text-[12px] leading-relaxed text-white/45">
   345→   345→                  The treatment on the world-page headline — chrome metal, neon glow, holographic sheen and friends.
   346→   346→                </p>
   347→   347→                <div className="mt-4">
   348→   348→                  <EffectPicker
   349→   349→                    value={theme.titleEffect || "none"}
   350→   350→                    onChange={(v) => setThemeKey("titleEffect", v === "none" ? null : v)}
   351→   351→                    accent={effAccent}
   352→   352→                    accent2={effAccent2}
   353→   353→                    fontFamily={displayStack ?? undefined}
   354→   354→                    sample={name || "AESTHETICA"}
   355→   355→                  />
   356→   356→                </div>
   357→   357→                <button
   358→   358→                  onClick={() => setTheme((t) => ({ ...t, fontDisplay: undefined, fontBody: undefined, epithetFont: undefined, titleEffect: undefined }))}
   359→   359→                  className="mt-4 flex items-center gap-1.5 text-[12px] text-white/50 hover:text-white"
   360→   360→                >
   361→   361→                  <RotateCcw className="h-3 w-3" /> Reset all typography to the world default
   362→   362→                </button>
   363→   363→              </section>
   364→   364→            </div>
   365→   365→          )}
   366→   366→
   367→   367→          {tab === "position" && (
   368→   368→            <div className="max-w-2xl space-y-7">
   369→   369→              <p className="text-[12.5px] leading-relaxed text-white/45">
   370→   370→                {def.kind === "custom"
   371→   371→                  ? "Layout of the world-page hero. Fine per-element positioning (X/Y/size/rotation of every scene layer) lives in the Scene builder tab — drag or nudge there."
   372→   372→                  : "Layout of the world-page hero: where the title block sits, how tall it breathes, and how wide the content column below runs."}
   373→   373→              </p>
   374→   374→              <div className="max-w-md">
   375→   375→                <SliderRow
   376→   376→                  label="Hero title size"
   377→   377→                  value={theme.titleScale ?? 1}
   378→   378→                  min={0.5}
   379→   379→                  max={2}
   380→   380→                  step={0.05}
   381→   381→                  format={(v) => `${Math.round(v * 100)}%`}
   382→   382→                  onReset={() => setThemeKey("titleScale", null)}
   383→   383→                  onChange={(v) => setTheme((t) => ({ ...t, titleScale: v }))}
   384→   384→                />
   385→   385→                <p className="mt-1 text-[11px] text-white/35">100% is the built-in size.</p>
   386→   386→              </div>
   387→   387→              <div>
   388→   388→                <label className="studio-label">Hero text alignment</label>
   389→   389→                <AlignPicker
   390→   390→                  value={(theme.heroAlign as AlignValue) || "center"}
